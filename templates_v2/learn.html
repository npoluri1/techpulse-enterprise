Quick Start: AI Agents in Python

Get started with agentic AI

Build a simple AI research agent using LangChain that can search the web and answer questions:

# pip install langchain langchain-openai chromadb

from langchain.agents import create_react_agent, AgentExecutor
from langchain.tools import Tool
from langchain_openai import ChatOpenAI
from langchain import hub

# Define tools
def search_tool(query: str) -> str:
    """Search for information online"""
    return f"Results for: {query}"

tools = [Tool(name="WebSearch", func=search_tool,
          description="Search the web")]

# Create agent
llm = ChatOpenAI(model="gpt-4", temperature=0)
prompt = hub.pull("hwchase17/react")
agent = create_react_agent(llm, tools, prompt)
agent_executor = AgentExecutor(agent=agent, tools=tools,
                               verbose=True)

response = agent_executor.invoke({
    "input": "What are the latest AI breakthroughs?"
})
print(response["output"])
